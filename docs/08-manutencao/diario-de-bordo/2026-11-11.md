## Verificador de links (recrie, não está no repositório)

Salve como `verificar-links.mjs` em qualquer pasta fora do repositório e rode `node verificar-links.mjs docs README.md CONTRIBUTING.md`:

```js
import fs from 'node:fs';
import path from 'node:path';
const files = [];
const walk = (p) => {
  const st = fs.statSync(p);
  if (st.isDirectory()) fs.readdirSync(p).forEach((f) => walk(path.join(p, f)));
  else if (p.endsWith('.md')) files.push(p);
};
process.argv.slice(2).forEach(walk);
let bad = 0, total = 0;
for (const f of files) {
  const txt = fs.readFileSync(f, 'utf8');
  for (const m of txt.matchAll(/\]\(([^)\s]+)\)/g)) {
    let t = m[1];
    if (/^(https?:|mailto:|#)/.test(t)) continue;
    t = t.split('#')[0];
    if (!t) continue;
    total++;
    if (!fs.existsSync(path.resolve(path.dirname(f), decodeURIComponent(t)))) { bad++; console.log(`${f}: ${m[1]}`); }
  }
}
console.log(`${files.length} arquivos, ${total} links, ${bad} quebrados`);
```
